$nerine.affection = 0
$nerine.diaryUnlocked = false

CreateCharacter(Nerine -i true -e true) 

SetLayerMedia(background "Fantasy City" -aud false -i true)

Nerine.SetPosition(0)

Nerine "You shouldn't have come back."

choice "How will you respond?"
{
    -Be gentle
        $el.affection += 15
        $nerine.affection += 5

    -Stay silent
        $el.affection += 5
        $nerine.affection += 10

    -Walk away
        $nerine.affection -= 5
}

Nerine.SetSprite(Nerine_Surprise)

if ($nerine.affection >= 10)
{
    $nerine.diaryUnlocked = true
    narrator "Nerine's diary has been unlocked!"
}

wait(2)

Nerine.SetSprite(Nerine_smile)
